# Practical 15: GCD and LCM Functions
def calculate_gcd(a, b):
    while b != 0:
        a, b = b, a % b
    return a
def calculate_lcm(a, b):
    if a == 0 or b == 0:
        return 0
    return abs(a * b) // calculate_gcd(a, b)

# Iterative Implementation
def factorial_iterative(n):
    result = 1
    for i in range(1, n + 1):
        result *= i
    return result
# Recursive Implementation
def factorial_recursive(n):
    if n == 0 or n == 1:  # Base Case
        return 1
    return n * factorial_recursive(n - 1)  # Recursive Step

# Practical 29: Polymorphism and Method Overriding
class Shape:
    def area(self):
        return "Area formula undefined for generic shape."
class Rectangle(Shape):
    def __init__(self, width, height):
        self.width = width
        self.height = height
    def area(self):  # Overriding
        return self.width * self.height
class Circle(Shape):
    def __init__(self, radius):
        self.radius = radius
    def area(self):  # Overriding
        return 3.14159 * (self.radius ** 2)
