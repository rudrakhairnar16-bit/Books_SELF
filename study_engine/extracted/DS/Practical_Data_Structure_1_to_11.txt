// Function to insert element at a specific index
int insertElement(int arr[], int *n, int pos, int val) {
    if (pos < 0 || pos > *n) {
        printf("Invalid position for insertion!\n");
        return 0;
    }
    for (int i = *n; i > pos; i--) {
        arr[i] = arr[i - 1];
    }
    arr[pos] = val;
    (*n)++;
    return 1;
}
// Function to delete element from a specific index
int deleteElement(int arr[], int *n, int pos) {
    if (pos < 0 || pos >= *n) {
        printf("Invalid position for deletion!\n");
        return 0;
    }
    int deletedVal = arr[pos];
    for (int i = pos; i < *n - 1; i++) {
        arr[i] = arr[i + 1];
    }
    (*n)--;
    return deletedVal;
}
// Function to perform linear search
int searchElement(int arr[], int n, int key) {
    for (int i = 0; i < n; i++) {
        if (arr[i] == key) {
            return i; // Return index where key is found
        }
    }
    return -1; // Return -1 if not found
}

int precedence(char ch) {
    switch (ch) {
        case '^': return 3;
        case '*':
        case '/': return 2;
        case '+':
        case '-': return 1;

// Function to convert Infix to Postfix
void infixToPostfix(char *infix, char *postfix) {
    int i = 0, j = 0;
    char symbol;
    while ((symbol = infix[i++]) != '\0') {
        if (isalnum(symbol)) {
            postfix[j++] = symbol;
        } else if (symbol == '(') {
            pushChar(symbol);
        } else if (symbol == ')') {
            while (top >= 0 && peekChar() != '(') {
                postfix[j++] = popChar();
            }
            popChar(); // Pop '('
        } else { // Operator encountered
            while (top >= 0 && precedence(peekChar()) >= precedence(symbol)) {
                postfix[j++] = popChar();
            }
            pushChar(symbol);
        }
    }
    // Pop remaining operators in stack
    while (top >= 0) {
        postfix[j++] = popChar();
    }
    postfix[j] = '\0';
}

// 1. Bubble Sort
void bubbleSort(int arr[], int n) {
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
}

// 3. Quick Sort Helper
int partition(int arr[], int low, int high) {
    int pivot = arr[high];
    int i = (low - 1);
    for (int j = low; j <= high - 1; j++) {
        if (arr[j] < pivot) {
            i++;
            int temp = arr[i];
            arr[i] = arr[j];
            arr[j] = temp;
        }
    }
    int temp = arr[i + 1];
    arr[i + 1] = arr[high];
    arr[high] = temp;
    return (i + 1);
}
void quickSort(int arr[], int low, int high) {
    if (low < high) {
        int pi = partition(arr, low, high);
        quickSort(arr, low, pi - 1);
        quickSort(arr, pi + 1, high);
    }
}
