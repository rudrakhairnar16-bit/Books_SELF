Implement a program to convert infix notation to postfix notation using stack. 
 
#include <stdio.h> 
#include <ctype.h> 
char stack[50]; 
int top = -1; 
int pre(char c) 
{ 
    if(c == '+' || c == '-') return 1; 
    if(c == '*' || c == '/') return 2; 
    if(c == '^') return 3; 
    return 0; 
} 
 
int main() 
{ 
    char in[50], out[50], c; 
    int i, j = 0; 
 
    printf("Enter infix expression: "); 
    scanf("%s", in); 
 
    for(i = 0; in[i] != '\0'; i++) 
    { 
        c = in[i]; 
 
        if(isalnum(c)) 
            out[j++] = c; 
 
        else if(c == '(') 
            stack[++top] = c; 
 
        else if(c == ')') 
        { 
            while(top != -1 && stack[top] != '(') 
                out[j++] = stack[top--]; 
            top--; 
        } 
 
        else 
        { 
            while(top != -1 && pre(stack[top]) >= pre(c)) 
                out[j++] = stack[top--]; 
 
            stack[++top] = c; 
        } 
    } 
 
    while(top != -1) 
        out[j++] = stack[top--]; 
    out[j] = '\0'; 
    printf("Postfix: %s", out); 
 
    return 0; 
}

Circular Queue 
 
#include <stdio.h> 
#define MAX 5 
 
int main() 
{ 
    int q[MAX], front = 0, rear = -1; 
    int i, del; 
    printf("Enter 5 values:\n"); 
 
    for(i = 0; i < MAX; i++) 
    { 
        rear = (rear + 1) % MAX;              Move rear circularly 
        scanf("%d", &q[rear]); 
    } 
    printf("Circular Queue: "); 
    for(i = 0; i < MAX; i++) 
        printf("%d ", q[(front + i) % MAX]);        
 
    printf("\nHow many values do you want to delete? "); 
    scanf("%d", &del); 
 
    if(del > MAX)                      deletion is greater than queue size 
        printf("Queue Underflow\n"); 
    else 
    { 
        for(i = 0; i < del; i++) 
        { 
            printf("Deleted: %d\n", q[front]); 
            front = (front + 1) % MAX;                          Move front forward 
        } 
        printf("Queue after deletion: "); 
        for(i = 0; i < MAX - del; i++)  
            printf("%d ", q[(front + i) % MAX]);              
    } 
 
    return 0; 
}

#include <stdio.h> 
 
void merge(int a[], int l, int m, int r) 
{ 
    int i = l, j = m + 1, k = 0; 
    int b[10]; 
 
    while(i <= m && j <= r)                            Comparing elements 
    { 
        if(a[i] < a[j])                                               Copies the smaller element into b. 
            b[k++] = a[i++]; 
        else 
            b[k++] = a[j++]; 
    } 
 
    while(i <= m)                                            Copy remaining elements 
        b[k++] = a[i++]; 
 
    while(j <= r) 
        b[k++] = a[j++]; 
 
    for(i = l, k = 0; i <= r; i++, k++) 
        a[i] = b[k]; 
} 
 
void mergeSort(int a[], int l, int r) 
{ 
    if(l < r) 
    { 
        int m = (l + r) / 2; 
 
        mergeSort(a, l, m); 
        mergeSort(a, m + 1, r); 
        merge(a, l, m, r); 
    } 
} 
 
int main() 
{ 
    int a[5], i; 
 
    printf("Enter 5 elements: "); 
    for(i = 0; i < 5; i++) 
        scanf("%d", &a[i]); 
 
    mergeSort(a, 0, 4); 
    printf("Sorted elements: "); 
    for(i = 0; i < 5; i++) 
        printf("%d ", a[i]); 
    return 0; 
}

#include <stdio.h>               
void quickSort(int a[], int low, int high) 
{ 
    int i, j, pivot, temp; 
    if(low < high) 
    { 
        pivot = a[low];                                        first element is selected as the pivot. 
        i = low + 1;                                             i starts from the element after the pivot. 
        j = high;                                                  j starts from the last element.  
        while(i <= j)                                           Continues comparing elements while i has not crossed j. 
        { 
            while(i <= high && a[i] <= pivot)                                   Move i 
                i++; 
            while(a[j] > pivot)                                                    Move j 
                j--; 
            if(i < j)                                                                          Check whether elements need swapping                                                              
            {   
                temp = a[i]; 
                a[i] = a[j]; 
                a[j] = temp; 
            } 
        } 
        temp = a[low];                                                                Put pivot in correct position 
        a[low] = a[j]; 
        a[j] = temp; 
        quickSort(a, low, j - 1);                                                   Sort left side 
        quickSort(a, j + 1, high);                                                Sort right side 
    } 
}
