-- Query 1: Equi/Inner Join (Customers who have both deposit and loan)
SELECT C.CNAME, C.CITY, D.AMOUNT AS DEPOSIT_AMT, B.AMOUNT AS LOAN_AMT
FROM CUSTOMERS C
JOIN DEPOSIT D ON C.CNAME = D.CNAME
JOIN BORROW B ON C.CNAME = B.CNAME;
-- Query 2: Left Outer Join (All customers and their deposit details)
SELECT C.CNAME, C.CITY, D.ACTNO, NVL(D.AMOUNT, 0) AS DEPOSIT_AMT
FROM CUSTOMERS C
LEFT OUTER JOIN DEPOSIT D ON C.CNAME = D.CNAME;
-- Query 3: Self Join (Finding employees in the same department)
SELECT E1.EMPNAME AS EMPLOYEE_1, E2.EMPNAME AS EMPLOYEE_2, E1.DEPTNO
FROM EMPLOYEE E1
JOIN EMPLOYEE E2 ON E1.DEPTNO = E2.DEPTNO
WHERE E1.EMPNO < E2.EMPNO;

    DBMS_OUTPUT.PUT_LINE('    HIGH-VALUE DEPOSITORS REPORT (BAL > 2000)    ');
    DBMS_OUTPUT.PUT_LINE('==================================================');
    
    -- 2. Open Cursor
    OPEN cur_depositors;
    
    -- 3. Fetch Records in a Loop
    LOOP
        FETCH cur_depositors INTO rec_dep;
        EXIT WHEN cur_depositors%NOTFOUND;
        
        DBMS_OUTPUT.PUT_LINE('ACT NO: ' || rec_dep.ACTNO || 
                             ' | NAME: ' || RPAD(rec_dep.CNAME, 10) || 
                             ' | BRANCH: ' || RPAD(rec_dep.BNAME, 10) || 
                             ' | AMT: Rs. ' || rec_dep.AMOUNT);
    END LOOP;
    
    DBMS_OUTPUT.PUT_LINE('--------------------------------------------------');
    DBMS_OUTPUT.PUT_LINE('TOTAL RECORDS PROCESSED: ' || cur_depositors%ROWCOUNT);
    DBMS_OUTPUT.PUT_LINE('==================================================');
    
    -- 4. Close Cursor
    CLOSE cur_depositors;
END;
